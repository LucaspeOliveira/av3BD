/****** Object:  Table [dbo].[tb_carros]    Script Date: 22/11/2025 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
CREATE TABLE [dbo].[tb_carros](
    [placa]			[varchar](10)	NOT NULL,
    [marca]			[varchar](100)	NOT NULL,
    [modelo]		[varchar](100)	NOT NULL,
    [cor]			[varchar](50)	NOT NULL,
    [ano]			[int]			NOT NULL,
    [combustivel]	[varchar](10)	NOT NULL,
    [km_rodados]	[decimal](10,2)	NOT NULL,
    [cambio]		[varchar](8)	NOT NULL,
    [status]		[varchar](10)	NOT NULL, -- status possíveis: DISPONIVEL, ALUGADO, REPARANDO.
    [categoria_id]	[bigint]		NOT NULL,
	PRIMARY KEY CLUSTERED(
		[placa] ASC
	) 
	ON [PRIMARY],
	FOREIGN KEY ([categoria_id]) REFERENCES [dbo].[tb_categorias]([id])
) 
ON [PRIMARY]
GO

CREATE TABLE [dbo].[tb_locacoes](
    [id]				[bigint] IDENTITY(1,1)	NOT NULL,
    [carro_placa]		[varchar](10)			NOT NULL,
    [cliente_cpf]		[varchar](12)			NOT NULL,
    [data_retirada]		[date]					NOT NULL,
    [dias_alugado]		[int]					NOT NULL,
    [data_devolucao]	[date]					NOT NULL,
    [tanque]			[varchar](8)			NOT NULL,
    [valor_total]		[decimal](10,2)			NOT NULL,
	PRIMARY KEY CLUSTERED(
		[id] ASC
	)
	ON [PRIMARY],
	FOREIGN KEY ([carro_placa]) REFERENCES [dbo].[tb_carros]([placa]),
	FOREIGN KEY ([cliente_cpf]) REFERENCES [dbo].[tb_clientes]([cpf])
)
ON [PRIMARY]
GO
